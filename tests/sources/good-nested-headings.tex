% good-nested-headings -- three heading levels with no text in between.
%
% The official template, unmodified, with headings stacked back to back.
% LaTeX's \@startsection combines the skips around stacked headings with
% \addvspace (the maximum, not the sum), and \flushbottom then shrinks that
% glue to fill the column, so the whitespace above a heading that directly
% follows another heading is both smaller than after body text and dependent
% on where the page breaks. None of that is the author's doing: it must not
% read as squeezing.
%
% Variants covered, in order: a plain three-level stack; four levels ending
% in a run-in \paragraph label; a stack whose parent heading TITLE WRAPS, so
% the line above carries no section number of its own; and two-digit section
% numbers (10.1.1).
%
% Expected checker outcome: PASS with zero findings.
% =============================================================================

\lipsum[1][1-5]

\lipsum[2-4]

\lipsum[5]

% four levels stacked, ending in a run-in \paragraph label
\subsection{Components}

\paragraph{Machines.}
\lipsum[6]

% a heading whose title wraps: the line above the \subsubsection is then a
% heading CONTINUATION line, carrying no section number of its own
\section{Evaluation}
\subsection{Setup of the Prototype Implementation and Its Runtime}
\subsubsection{Hardware}
\lipsum[7]
